\begin{theorem}\label{ZL for K3 carpets}
 Let $\widetilde{Y} \hookrightarrow \mathbb{P}^M$ be a $K3$ carpet embedded by the complete linear series of a very ample line bundle $\widetilde{H}$ such that the embedding induced on the reduced part is $Y = \mathbb{F}_e \hookrightarrow \mathbb{P}^N$ embedded by the complete linear series of a very ample line bundle $H = C_0+bf$. If $b \geq e+3$, then $\alpha(\widetilde Y) = h^0(\mathcal{O}_Y(-H-2K_Y))$.
 \end{theorem}
